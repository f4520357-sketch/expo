
import React, { useState } from "react";
import {
  View, Text, TouchableOpacity, ScrollView,
  TextInput, StyleSheet, SafeAreaView
} from "react-native";

const RED = "#F12B36";
const BG = "#090909";
const CARD = "#191919";
const WHITE = "#FFFFFF";

export default function App() {
  const [page, setPage] = useState("الرئيسية");
  const [name, setName] = useState("");
  const [players, setPlayers] = useState(["جيكا", "محمد", "أحمد", "محمود"]);
  const [matches, setMatches] = useState([]);
  const [teams, setTeams] = useState([]);
  const [cost, setCost] = useState("");
  const [pitch, setPitch] = useState("");

  const addPlayer = () => {
    if (name.trim() && !players.includes(name.trim())) {
      setPlayers([...players, name.trim()]);
      setName("");
    }
  };

  const addMatch = () => {
    if (!pitch.trim()) return;
    setMatches([...matches, {
      pitch, cost: cost || "0", date: new Date().toLocaleDateString("ar-EG")
    }]);
    setPitch("");
    setCost("");
  };

  const makeTeams = () => {
    setTeams([
      { captain: players[0] || "القائد الأول", members: players.filter((_,i)=>i%2===0) },
      { captain: players[1] || "القائد الثاني", members: players.filter((_,i)=>i%2===1) }
    ]);
    setPage("الفرق");
  };

  const menu = ["الرئيسية", "اللاعبين", "المباريات", "الفرق", "المدفوعات"];

  return (
    <SafeAreaView style={s.container}>
      <View style={s.header}>
        <Text style={s.logo}>👑 دوري الصحاب ⚽</Text>
        <Text style={s.subtitle}>أهلاً بيك يا جيكا</Text>
      </View>

      <ScrollView contentContainerStyle={s.content}>
        {page === "الرئيسية" && <>
          <Text style={s.title}>لوحة الدوري</Text>
          <View style={s.stats}>
            <Stat label="اللاعبين" value={players.length}/>
            <Stat label="المباريات" value={matches.length}/>
          </View>
          <Card title="أهلاً بيك في دوري الصحاب">
            <Text style={s.text}>كل تفاصيل الدوري في مكان واحد.</Text>
          </Card>
        </>}

        {page === "اللاعبين" && <>
          <Text style={s.title}>إدارة اللاعبين</Text>
          <TextInput style={s.input} placeholder="اسم اللاعب الجديد"
            placeholderTextColor="#999" value={name} onChangeText={setName}/>
          <Btn label="إضافة لاعب" onPress={addPlayer}/>
          {players.map((p,i)=><Card key={i} title={"⚽ "+p}>
            <Text style={s.text}>لاعب في دوري الصحاب</Text>
          </Card>)}
        </>}

        {page === "المباريات" && <>
          <Text style={s.title}>المباريات والحجوزات</Text>
          <TextInput style={s.input} placeholder="اسم الملعب"
            placeholderTextColor="#999" value={pitch} onChangeText={setPitch}/>
          <TextInput style={s.input} placeholder="تكلفة الملعب"
            placeholderTextColor="#999" keyboardType="numeric"
            value={cost} onChangeText={setCost}/>
          <Btn label="إنشاء مباراة" onPress={addMatch}/>
          {matches.map((m,i)=><Card key={i} title={"🏟️ "+m.pitch}>
            <Text style={s.text}>التكلفة: {m.cost} جنيه</Text>
            <Text style={s.text}>التاريخ: {m.date}</Text>
          </Card>)}
        </>}

        {page === "الفرق" && <>
          <Text style={s.title}>تقسيم الفرق</Text>
          <Btn label="تكوين فريقين تجريبيًا" onPress={makeTeams}/>
          {teams.map((t,i)=><Card key={i} title={"الفريق "+(i+1)}>
            <Text style={s.text}>القائد: {t.captain}</Text>
            {t.members.map((p,j)=><Text key={j} style={s.text}>• {p}</Text>)}
          </Card>)}
        </>}

        {page === "المدفوعات" && <>
          <Text style={s.title}>المدفوعات</Text>
          <Card title="حساب تكلفة الملعب">
            {matches.map((m,i)=><Text key={i} style={s.text}>
              {m.pitch}: {m.cost} جنيه
            </Text>)}
            <Text style={s.text}>عدد اللاعبين: {players.length}</Text>
            <Text style={s.text}>تقسيم التكلفة يحتاج تأكيد المشاركين.</Text>
          </Card>
        </>}
      </ScrollView>

      <View style={s.nav}>
        {menu.map(item=><TouchableOpacity key={item}
          onPress={()=>setPage(item)} style={s.navItem}>
          <Text style={[s.navText,page===item&&s.active]}>{item}</Text>
        </TouchableOpacity>)}
      </View>
    </SafeAreaView>
  );
}

function Stat({label,value}) {
  return <View style={s.stat}>
    <Text style={s.number}>{value}</Text>
    <Text style={s.text}>{label}</Text>
  </View>;
}

function Card({title,children}) {
  return <View style={s.card}>
    <Text style={s.cardTitle}>{title}</Text>
    {children}
  </View>;
}

function Btn({label,onPress}) {
  return <TouchableOpacity style={s.button} onPress={onPress}>
    <Text style={s.buttonText}>{label}</Text>
  </TouchableOpacity>;
}

const s = StyleSheet.create({
  container:{flex:1,backgroundColor:BG},
  header:{padding:22,backgroundColor:CARD,borderBottomWidth:2,borderBottomColor:RED},
  logo:{fontSize:25,fontWeight:"bold",color:RED,textAlign:"center"},
  subtitle:{color:WHITE,textAlign:"center",marginTop:8},
  content:{padding:16,paddingBottom:30},
  title:{fontSize:23,color:RED,fontWeight:"bold",marginBottom:18,textAlign:"right"},
  stats:{flexDirection:"row",gap:12,marginBottom:15},
  stat:{flex:1,backgroundColor:CARD,padding:20,borderRadius:12,alignItems:"center"},
  number:{fontSize:28,color:RED,fontWeight:"bold"},
  text:{color:WHITE,fontSize:15,marginTop:8,textAlign:"right"},
  card:{backgroundColor:CARD,padding:16,borderRadius:12,marginBottom:12,borderRightWidth:3,borderRightColor:RED},
  cardTitle:{color:WHITE,fontSize:18,fontWeight:"bold",textAlign:"right"},
  input:{backgroundColor:CARD,color:WHITE,padding:14,borderRadius:10,marginBottom:12,textAlign:"right"},
  button:{backgroundColor:RED,padding:15,borderRadius:10,alignItems:"center",marginBottom:16},
  buttonText:{color:WHITE,fontWeight:"bold",fontSize:16},
  nav:{flexDirection:"row",justifyContent:"space-around",backgroundColor:CARD,paddingVertical:13,borderTopWidth:1,borderTopColor:RED},
  navItem:{paddingHorizontal:3},
  navText:{color:"#AAA",fontSize:11},
  active:{color:RED,fontWeight:"bold"}
});
